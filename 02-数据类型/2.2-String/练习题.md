def middle(s):
    n = len(s)
    if n == 0:
        return ""

# 题目 1
s = "hello world hello"
print(s.find("hello"))      # 0
print(s.rfind("hello"))     # 12
print(s.find("hello", 1))   # 12
print(s.find("xyz"))        # -1（不抛异常）

# 题目 2
s = "mississippi"
print(s.index("s"))     # 2
print(s.count("s"))     # 4
print(s.count("ss"))    # 2（不重叠计数：位置 2-3, 5-6）
# s.index("x")  # ValueError: substring not found

# 题目 3
print("hello world".replace("world", "Python"))   # hello Python
print("a,b,c,d".replace(",", " ", 2))             # a b c,d
print("  hello  ".replace(" ", ""))               # hello

# 题目 4
print("hello.py".endswith(".py"))   # True

print("image.png".endswith((".png", ".jpg", ".jpeg")))   # True

print("https://x.com".startswith(("http://", "https://")))   # True
print("ftp://x.com".startswith(("http://", "https://")))     # False

# 题目 5
def find_all(s, sub):
    if not sub:
        return []
    indices = []
    start = 0
    while True:
        idx = s.find(sub, start)
        if idx == -1:
            break
        indices.append(idx)
        start = idx + 1   # 允许重叠；若要不重叠用 idx + len(sub)
    return indices

print(find_all("hello world hello", "hello"))   # [0, 12]
print(find_all("mississippi", "ss"))            # [2, 5]
print(find_all("abc", "x"))                     # []

# 列表推导式版本
def find_all_v2(s, sub):
    return [i for i in range(len(s)) if s.startswith(sub, i)]

print(find_all_v2("mississippi", "ss"))   # [2, 5]
```

# 题目 1
print("hello".isalpha())      # True
print("hello123".isalnum())   # True

# 题目 4
def is_valid_age(s):
    if not s.isdigit():
        return False
    n = int(s)
    return 1 <= n <= 150

print(is_valid_age("20"))    # True
print(is_valid_age("0"))     # False
print(is_valid_age("200"))   # False
print(is_valid_age("abc"))   # False
print(is_valid_age("-5"))    # False - isdigit 不接受负号

# 题目 5
def is_strong_password(pwd):
    if len(pwd) < 8:
        return False
    has_upper = any(c.isupper() for c in pwd)
    has_lower = any(c.islower() for c in pwd)
    has_digit = any(c.isdigit() for c in pwd)
    return has_upper and has_lower and has_digit

print(is_strong_password("Abc12345"))   # True
print(is_strong_password("abc12345"))   # False - 无大写
print(is_strong_password("Abcde"))      # False - 太短
```

b_utf8 = s.encode("utf-8")

    print(f"错误: {e}")

# 题目 3
print(ord('A'))      # 65

# 大小写转换：A(65) ↔ a(97)，差 32
def to_lower(c):
    if 'A' <= c <= 'Z':
        return chr(ord(c) + 32)
    return c

def to_upper(c):
    if 'a' <= c <= 'z':
        return chr(ord(c) - 32)

    for i in range(1, len(s)):

            count += 1
        else:

        cur = values[c]
        if cur < prev:
            total -= cur
        else:
            total += cur
        prev = cur
    return total

def longest_palindrome(s):
    if not s:
        return ""
    n = len(s)
    start, max_len = 0, 1
    # 中心扩展
    for i in range(n):
        # 奇数长度
        l, r = i, i
        while l >= 0 and r < n and s[l] == s[r]:
            if r - l + 1 > max_len:
                start = l
                max_len = r - l + 1
            l -= 1
            r += 1
        # 偶数长度
        l, r = i, i + 1
        while l >= 0 and r < n and s[l] == s[r]:
            if r - l + 1 > max_len:
                start = l
                max_len = r - l + 1
            l -= 1
            r += 1
    return s[start:start + max_len]

print(longest_palindrome("cbbd"))    # bb

def add_strings(num1, num2):
    i, j = len(num1) - 1, len(num2) - 1
    carry = 0
    result = []
    while i >= 0 or j >= 0 or carry:
        d1 = ord(num1[i]) - ord('0') if i >= 0 else 0
        d2 = ord(num2[j]) - ord('0') if j >= 0 else 0
        total = d1 + d2 + carry
        carry = total // 10
        result.append(str(total % 10))
        i -= 1
        j -= 1
    return "".join(reversed(result))

print(add_strings("999", "1"))     # 1000
print(add_strings("0", "0"))       # 0

def caesar(text, shift):
    result = []
    for c in text:
        if c.isupper():
            result.append(chr((ord(c) - 65 + shift) % 26 + 65))
        elif c.islower():
            result.append(chr((ord(c) - 97 + shift) % 26 + 97))
        else:
            result.append(c)
    return "".join(result)

### 学习建议

1. **循序渐进**：从基础题开始，逐步挑战综合题
2. **动手实践**：每道题都要自己运行验证
3. **理解原理**：字符串不可变是核心，理解所有方法都返回新字符串
4. **多种方法**：尝试用不同方法解决同一问题
5. **性能意识**：循环拼接用 join，格式化用 f-string
6. **编码意识**：处理中文时注意 UTF-8/GBK 区别

### 常见模式总结

1. **反转**：`s[::-1]` 或 `"".join(reversed(s))`
2. **去重保序**：`"".join(dict.fromkeys(s))`
3. **回文判断**：预处理后 `s == s[::-1]`
4. **大小写不敏感比较**：`a.casefold() == b.casefold()`
5. **拼接优化**：`sep.join(iterable)`
6. **格式化首选**：f-string
7. **正则提取**：`re.findall(pattern, s)`
