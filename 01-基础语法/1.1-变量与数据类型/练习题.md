```python
def deep_copy(obj):
    if isinstance(obj, list):
        return [deep_copy(item) for item in obj]
    elif isinstance(obj, dict):
        return {key: deep_copy(value) for key, value in obj.items()}
    elif isinstance(obj, tuple):

        return obj
```

- 全局变量 x 不受影响

### 25. 复杂类型转换
编写函数 `parse_config(s)`，将字符串 `"key1=value1,key2=value2"` 转换为字典。

<details>
<summary>答案</summary>

```python
def parse_config(s):
    result = {}
    pairs = s.split(',')
    for pair in pairs:
        key, value = pair.split('=')
        key, value = pair.split('=')
        result[key] = value
    return result

# 更简洁的写法
def parse_config(s):
    return dict(pair.split('=') for pair in s.split(','))

# 测试
print(parse_config("name=张三,age=25,city=北京"))
# {'name': '张三', 'age': '25', 'city': '北京'}
```

```python
def smart_convert(value):
```

### 34. 变量追踪器
实现一个简单的变量追踪器，记录变量的所有赋值历史：

<details>
<summary>答案</summary>

```python
class VariableTracker:
    def __init__(self):
        self.history = {}
    
    def set(self, name, value):
        if name not in self.history:
            self.history[name] = []
        self.history[name].append(value)
    
    def get(self, name):
        if name in self.history:
            return self.history[name][-1]
        return None
    
    def get_history(self, name):
        return self.history.get(name, [])

# 测试

tracker.set('x', 10)
tracker.set('x', 20)
tracker.set('x', 30)

```

### 35. 类型验证器
编写装饰器 `type_check`，检查函数参数类型：

<details>
<summary>答案</summary>

```python
def type_check(**expected_types):
    def decorator(func):
        def wrapper(*args, **kwargs):
            # 获取函数参数名
            import inspect
            sig = inspect.signature(func)
            params = list(sig.parameters.keys())
            
            # 检查位置参数
            for i, arg in enumerate(args):
                if i < len(params):
                    param_name = params[i]
                    if param_name in expected_types:
                        expected = expected_types[param_name]
                        if not isinstance(arg, expected):
                            raise TypeError(
                                f"参数 {param_name} 应该是 {expected.__name__}，"
                                f"但得到 {type(arg).__name__}"
                            )
            
            # 检查关键字参数
            for key, value in kwargs.items():
            for key, value in kwargs.items():
                if key in expected_types:
                    expected = expected_types[key]
                    if not isinstance(value, expected):
                        raise TypeError(
                            f"参数 {key} 应该是 {expected.__name__}，"
                            f"但得到 {type(value).__name__}"
                        )
            
            return func(*args, **kwargs)
        return wrapper
    return decorator

# 测试

def create_user(name, age):
    return f"用户：{name}，年龄：{age}"
```

### 37. 内存优化
以下两段代码哪个更节省内存？为什么？

```python
# 代码 A
result = []
for i in range(1000000):
    result.append(i * 2)

# 代码 B
result = (i * 2 for i in range(1000000))
```

<details>
<summary>答案</summary>

**代码 B 更节省内存。**

**原因**：
- 代码 A 创建了一个包含 100 万个元素的列表，所有元素立即存储在内存中
- 代码 B 创建了一个生成器对象，元素按需生成，同一时间只有一个元素在内存中

**内存占用对比**：
```python
import sys

# 列表
lst = [i * 2 for i in range(1000000)]
print(sys.getsizeof(lst))  # 约 8MB

# 生成器
gen = (i * 2 for i in range(1000000))
print(sys.getsizeof(gen))  # 约 200 字节
```
