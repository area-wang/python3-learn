### 29. 密码验证
编写密码验证程序，要求：
- 长度至少 8 位
- 包含数字和字母
- 最多尝试 3 次

<details>
<summary>答案</summary>

```python
def validate_password(password):
    if len(password) < 8:
        return False, "密码至少 8 位"
    
    has_digit = any(c.isdigit() for c in password)
    has_alpha = any(c.isalpha() for c in password)
    
    if not has_digit:
        return False, "密码必须包含数字"
    if not has_alpha:
        return False, "密码必须包含字母"
    
    return True, "密码符合要求"

attempts = 0
max_attempts = 3

while attempts < max_attempts:
    password = input("请设置密码: ")
    valid, message = validate_password(password)
    
    if valid:
        print(message)
        break
    else:
        attempts += 1
        remaining = max_attempts - attempts
        print(f"{message}，还有 {remaining} 次机会")
else:
    print("超过最大尝试次数")
```
</details>

### 30. BMI 计算器
编写 BMI 计算器，输入体重和身高，输出 BMI 和健康建议。

<details>
<summary>答案</summary>

```python
def calculate_bmi():
    print("=== BMI 计算器 ===")
    
    try:
        weight = float(input("请输入体重(kg): "))
        height = float(input("请输入身高(m): "))
        
        if weight <= 0 or height <= 0:
            print("体重和身高必须大于 0")
            return
        
        bmi = weight / (height ** 2)
        
        # 判断健康状况
        if bmi < 18.5:
            category = "偏瘦"
            advice = "建议增加营养摄入"
        elif bmi < 24:
            category = "正常"
            advice = "保持良好习惯"
        elif bmi < 28:
            category = "偏胖"
            advice = "建议适当运动"
        else:
            category = "肥胖"
            advice = "建议咨询医生"
        
        print(f"\nBMI: {bmi:.2f}")
        print(f"分类: {category}")
        print(f"建议: {advice}")
        
    except ValueError:
        print("请输入有效的数字")

calculate_bmi()
```

### 31. 猜数字游戏

- 记录猜测次数

```python
import random

def guess_number():
    secret = random.randint(1, 100)
    attempts = 0
    
    print("我想了一个 1 到 100 之间的数字，请猜猜是多少？")
    
    while True:
        try:
            guess = int(input("请输入你的猜测: "))
            attempts += 1
            
            if guess < 1 or guess > 100:
                print("请输入 1 到 100 之间的数字")
                continue
            
            if guess == secret:
                print(f"恭喜你，猜对了！用了 {attempts} 次")
                break
            elif guess < secret:
                print("太小了")
            else:
                print("太大了")
        
        except ValueError:
            print("请输入有效的数字")

guess_number()
```

```python
def show_menu():

                    print("请输入有效的数字")
```

### 33. 购物清单
编写购物清单程序，支持添加、删除、显示商品及总价。

<details>
<summary>答案</summary>

```python
def shopping_list():
    cart = {}  # {商品名: 价格}
    
    while True:
        print("\n=== 购物清单 ===")
        print("1. 添加商品")
        print("2. 删除商品")
        print("3. 查看清单")
        print("4. 结算")
        print("5. 退出")
        
        choice = input("请选择 (1-5): ").strip()
        
        if choice == "1":
            name = input("商品名称: ").strip()
            try:
                price = float(input("价格: "))
                if price < 0:
                    print("价格不能为负数")
                    continue
                cart[name] = price
                print(f"已添加: {name} - ¥{price:.2f}")
            except ValueError:
                print("请输入有效的价格")
        
        elif choice == "2":
            if cart:
                print("当前商品:", ", ".join(cart.keys()))
                name = input("要删除的商品: ").strip()
                if name in cart:
                    del cart[name]
                    print(f"已删除: {name}")
                else:
                    print("商品不存在")
            else:
                print("清单为空")
        
        elif choice == "3":
            if cart:
                print(f"\n{'商品':<10} {'价格':<10}")
                print("-" * 20)
                for name, price in cart.items():
                    print(f"{name:<10} ¥{price:<9.2f}")
                total = sum(cart.values())
                print("-" * 20)
                print(f"{'总计':<10} ¥{total:.2f}")
            else:
                print("清单为空")
        
        elif choice == "4":
            if cart:
                total = sum(cart.values())
                print(f"\n总金额: ¥{total:.2f}")
                confirm = input("确认结算？(y/n): ").lower()
                if confirm == "y":
                    print("结算完成，感谢惠顾！")
                    cart.clear()
            else:
                print("清单为空")
        
        elif choice == "5":
            print("再见！")
            break
        
        else:
            print("无效的选择")

shopping_list()
```

```python
def generate_report():
```

### 35. 命令行工具
编写一个简单的命令行工具，支持多种命令。

<details>
<summary>答案</summary>

```python
def calculator(expr):
    """简单计算器"""
    try:
        result = eval(expr)
        return f"结果: {result}"
    except:
        return "计算错误"

def reverse_text(text):
    """反转文本"""
    return text[::-1]

def count_words(text):
    """统计单词数"""
    words = text.split()
    return f"单词数: {len(words)}"

def main():
    print("=== 命令行工具 ===")
    print("命令列表:")
    print("  calc <表达式>  - 计算器")
    print("  reverse <文本> - 反转文本")
    print("  count <文本>   - 统计单词数")
    print("  help          - 显示帮助")
    print("  exit          - 退出")
    print()
    
    while True:
        command = input("> ").strip()
        
        if not command:
            continue
        
        parts = command.split(maxsplit=1)
        cmd = parts[0].lower()
        
        if cmd == "exit":
            print("再见！")
            break
        
        elif cmd == "help":
            print("calc <表达式>  - 计算器")
            print("reverse <文本> - 反转文本")
            print("count <文本>   - 统计单词数")
        
        elif cmd == "calc":
            if len(parts) > 1:
                print(calculator(parts[1]))
            else:
                print("用法: calc <表达式>")
        
        elif cmd == "reverse":
            if len(parts) > 1:
                print(reverse_text(parts[1]))
            else:
                print("用法: reverse <文本>")
        
        elif cmd == "count":
            if len(parts) > 1:
                print(count_words(parts[1]))
            else:
                print("用法: count <文本>")
        
        else:
            print(f"未知命令: {cmd}")
            print("输入 'help' 查看帮助")

main()
```
