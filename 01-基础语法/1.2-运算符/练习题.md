```python
    # 如果最后一位是1，则为奇数

# 题目 2
def swap(a, b):
    a = a ^ b
    b = a ^ b  # b = (a^b) ^ b = a
    a = a ^ b  # a = (a^b) ^ a = b
    return a, b

a, b = swap(5, 10)
print(a, b)  # 10 5

# 题目 3
def power_of_2(n):
    return 1 << n

# 题目 4
num = 5  # 0101

# 获取第2位的值
bit_2 = (num >> 2) & 1
print(f"第2位的值：{bit_2}")  # 1

# 将第1位设置为1
num = num | (1 << 1)
print(f"设置第1位为1：{num} ({bin(num)})")  # 7 (0111)

# 将第2位设置为0
num = 5
num = num & ~(1 << 2)
print(f"设置第2位为0：{num} ({bin(num)})")  # 1 (0001)

# 翻转第0位
num = 5
num = num ^ (1 << 0)
print(f"翻转第0位：{num} ({bin(num)})")  # 4 (0100)

# 题目 5
def find_unique(nums):
    result = 0
    for num in nums:
        result ^= num
    return result

print(find_unique([2, 3, 2, 4, 4]))  # 3
print(find_unique([1, 1, 2, 2, 3]))  # 3

# 原理：相同的数异或为0，任何数与0异或等于自己
# 2 ^ 3 ^ 2 ^ 4 ^ 4 = (2^2) ^ 3 ^ (4^4) = 0 ^ 3 ^ 0 = 3
```

```python
        return f"两个复数根：x1 = {real:.2f} + {imag:.2f}i, x2 = {real:.2f} - {imag:.2f}i"
```

```python
def is_power_of_2(n):
    """
    原理：2的幂的二进制只有一个1
    n & (n-1) 会把最右边的1变成0
    如果n是2的幂，则n & (n-1) = 0
    """
    if n <= 0:
        return False
    return (n & (n - 1)) == 0

print(is_power_of_2(18))  # False (10010)

# 原理解释：

# 18: 10010, 17: 10001, 18 & 17 = 10000 ≠ 0 ✗

# 题目 2：位运算实现加法
def add(a, b):
    """
    原理：
    - 异或(^)：不考虑进位的加法
    - 与(&)后左移：计算进位
    """
    MAX = 0x7FFFFFFF
    mask = 0xFFFFFFFF
    
    while b != 0:
        # 计算不带进位的和
        sum_without_carry = (a ^ b) & mask
        # 计算进位
        carry = ((a & b) << 1) & mask
        
        a = sum_without_carry
        b = carry
    
    return a if a <= MAX else ~(a ^ mask)

print(add(5, 3))   # 8
print(add(10, 7))  # 17

def count_ones(n):
    """Brian Kernighan算法"""
    count = 0
    while n:
        n &= (n - 1)  # 消除最右边的1
        count += 1
    return count

# 题目 4：表达式计算器（简化版）
def calculate(expression):
    """支持加减乘除和括号"""
    expression = expression.replace(' ', '')
    
    def helper(s, start):
        stack = []
        num = 0
        op = '+'
        
        for i in range(start, len(s)):
            c = s[i]
            
            if c.isdigit():
                num = num * 10 + int(c)
            
            if c == '(':
                num, next_i = helper(s, i + 1)
                i = next_i
            
            if c in '+-*/' or c == ')' or i == len(s) - 1:
                if op == '+':
                    stack.append(num)
                elif op == '-':
                    stack.append(-num)
                elif op == '*':
                    stack.append(stack.pop() * num)
                elif op == '/':
                    stack.append(int(stack.pop() / num))
                
                op = c
                num = 0
                
                if c == ')':
                    return sum(stack), i
        
        return sum(stack), len(s)
    
    result, _ = helper(expression, 0)
    return result

print(calculate("2 + 3 * 4"))        # 14
print(calculate("(2 + 3) * 4"))      # 20
print(calculate("10 / 2 + 3"))       # 8

# 题目 5：位操作工具类
class BitOperations:
    @staticmethod
    def get_bit(num, i):
        """获取第i位的值（从右往左，从0开始）"""
        return (num >> i) & 1
    
    @staticmethod
    def set_bit(num, i):
        """将第i位设置为1"""
        return num | (1 << i)
    
    @staticmethod
    def clear_bit(num, i):
        """将第i位设置为0"""
        return num & ~(1 << i)
    
    @staticmethod
    def toggle_bit(num, i):
        """翻转第i位"""
        return num ^ (1 << i)
    
    @staticmethod
    def update_bit(num, i, value):
        """将第i位更新为value（0或1）"""
        mask = ~(1 << i)  # 先清除第i位
        return (num & mask) | (value << i)
    
    @staticmethod
    def is_even(num):
        """判断是否为偶数"""
        return (num & 1) == 0
    
    @staticmethod
    def multiply_by_2(num):
        """乘以2"""
        return num << 1
    
    @staticmethod
    def divide_by_2(num):
        """除以2"""
        return num >> 1
```
