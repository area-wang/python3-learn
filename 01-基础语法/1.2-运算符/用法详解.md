# Python 运算符详解

## 1. 算术运算符

### 1.1 基本算术运算符

| 运算符 | 名称 | 示例 | 结果 |
|--------|------|------|------|
| + | 加法 | 5 + 3 | 8 |
| - | 减法 | 5 - 3 | 2 |
| * | 乘法 | 5 * 3 | 15 |
| / | 除法 | 5 / 2 | 2.5 |
| // | 整除 | 5 // 2 | 2 |
| % | 取模(余数) | 5 % 2 | 1 |
| ** | 幂运算 | 5 ** 2 | 25 |

### 1.2 详细说明

#### 加法运算符 (+)
```python
# 数字相加
a = 10 + 5  # 15

# 字符串拼接
str1 = "Hello" + " " + "World"  # "Hello World"

# 列表合并
list1 = [1, 2] + [3, 4]  # [1, 2, 3, 4]

# 元组合并
tuple1 = (1, 2) + (3, 4)  # (1, 2, 3, 4)
```

#### 减法运算符 (-)
```python
a = 10 - 5  # 5
b = 3.5 - 1.2  # 2.3

# 负号
c = -5  # -5
d = -(10 - 3)  # -7
```

#### 乘法运算符 (*)
```python
# 数字相乘
a = 5 * 3  # 15

# 字符串重复
str1 = "Python" * 3  # "PythonPythonPython"

# 列表重复
list1 = [1, 2] * 3  # [1, 2, 1, 2, 1, 2]
```

#### 除法运算符 (/)
```python
# 总是返回浮点数
a = 10 / 3  # 3.3333333333333335
b = 10 / 2  # 5.0 (注意：即使整除也返回float)
c = 7 / 2   # 3.5
```

#### 整除运算符 (//)
```python
# 向下取整
a = 10 // 3  # 3
b = 10 // 2  # 5
c = -10 // 3  # -4 (向负无穷方向取整)
d = 10.5 // 2  # 5.0
```

#### 取模运算符 (%)
```python
# 返回除法的余数
a = 10 % 3  # 1
b = 10 % 2  # 0
c = 7 % 4   # 3

# 负数取模
d = -10 % 3  # 2 (Python中结果符号与除数相同)
e = 10 % -3  # -2

# 实际应用：判断奇偶数
num = 7
if num % 2 == 0:
    print("偶数")
else:
    print("奇数")

# 实际应用：循环索引
for i in range(10):
    index = i % 3  # 结果循环在 0, 1, 2
```

#### 幂运算符 (**)
```python
# 计算乘方
a = 2 ** 3   # 8 (2的3次方)
b = 5 ** 2   # 25 (5的平方)
c = 10 ** 0  # 1 (任何数的0次方都是1)

# 计算开方
d = 9 ** 0.5   # 3.0 (平方根)
e = 27 ** (1/3)  # 3.0 (立方根)

# 负指数
f = 2 ** -1  # 0.5 (2的-1次方 = 1/2)
```

## 2. 比较运算符

### 2.1 比较运算符表

| 运算符 | 名称 | 示例 | 结果 |
|--------|------|------|------|
| == | 等于 | 5 == 5 | True |
| != | 不等于 | 5 != 3 | True |
| > | 大于 | 5 > 3 | True |
| < | 小于 | 5 < 3 | False |
| >= | 大于等于 | 5 >= 5 | True |
| <= | 小于等于 | 5 <= 3 | False |

### 2.2 详细说明

```python
# 数字比较
a = 10
b = 20
print(a == b)  # False
print(a != b)  # True
print(a < b)   # True
print(a > b)   # False
print(a <= 10) # True
print(a >= 10) # True

# 字符串比较（按字典序）
print("apple" < "banana")  # True
print("Apple" < "apple")   # True (大写字母ASCII值小)
print("abc" == "abc")      # True

# 列表比较（逐元素比较）
print([1, 2, 3] < [1, 2, 4])  # True
print([1, 2] < [1, 2, 3])     # True (长度短的小)

# 注意：== 和 is 的区别
list1 = [1, 2, 3]
list2 = [1, 2, 3]
list3 = list1

print(list1 == list2)  # True (值相同)
print(list1 is list2)  # False (不是同一个对象)
print(list1 is list3)  # True (是同一个对象)
```

## 3. 赋值运算符

### 3.1 赋值运算符表

| 运算符 | 示例 | 等价于 |
|--------|------|--------|
| = | x = 5 | x = 5 |
| += | x += 3 | x = x + 3 |
| -= | x -= 3 | x = x - 3 |
| *= | x *= 3 | x = x * 3 |
| /= | x /= 3 | x = x / 3 |
| //= | x //= 3 | x = x // 3 |
| %= | x %= 3 | x = x % 3 |
| **= | x **= 3 | x = x ** 3 |

### 3.2 详细说明

```python
# 基本赋值
x = 10
print(x)  # 10

# 加法赋值
x = 10
x += 5  # x = x + 5
print(x)  # 15

# 减法赋值
x = 10
x -= 3  # x = x - 3
print(x)  # 7

# 乘法赋值
x = 10
x *= 2  # x = x * 2
print(x)  # 20

# 除法赋值
x = 10
x /= 3  # x = x / 3
print(x)  # 3.3333...

# 整除赋值
x = 10
x //= 3  # x = x // 3
print(x)  # 3

# 取模赋值
x = 10
x %= 3  # x = x % 3
print(x)  # 1

# 幂赋值
x = 2
x **= 3  # x = x ** 3
print(x)  # 8

# 多重赋值
a = b = c = 10
print(a, b, c)  # 10 10 10

# 序列解包赋值
x, y, z = 1, 2, 3
print(x, y, z)  # 1 2 3

# 交换变量
a, b = 10, 20
a, b = b, a  # 交换
print(a, b)  # 20 10
```

## 4. 逻辑运算符

### 4.1 逻辑运算符表

| 运算符 | 名称 | 示例 | 说明 |
|--------|------|------|------|
| and | 逻辑与 | x and y | 两者都为True时返回True |
| or | 逻辑或 | x or y | 至少一个为True时返回True |
| not | 逻辑非 | not x | 取反 |

### 4.2 详细说明

#### and 运算符
```python
# 基本用法
print(True and True)   # True
print(True and False)  # False
print(False and True)  # False
print(False and False) # False

# 实际应用
age = 25
has_license = True
if age >= 18 and has_license:
    print("可以开车")

# 短路求值：如果左边为False，右边不执行
x = 0
result = (x != 0) and (10 / x > 1)  # 不会报错，因为左边已经是False

# 返回值特性：返回第一个False值或最后一个True值
print(5 and 10)   # 10
print(0 and 10)   # 0
print(5 and 0)    # 0
```

#### or 运算符
```python
# 基本用法
print(True or True)    # True
print(True or False)   # True
print(False or True)   # True
print(False or False)  # False

# 实际应用
role = "admin"
if role == "admin" or role == "moderator":
    print("有管理权限")

# 短路求值：如果左边为True，右边不执行
x = 10
result = (x == 10) or (10 / 0 > 1)  # 不会报错

# 返回值特性：返回第一个True值或最后一个False值
print(5 or 10)    # 5
print(0 or 10)    # 10
print(0 or False) # False

# 默认值设置
name = ""
display_name = name or "匿名用户"  # 如果name为空，使用默认值
print(display_name)  # "匿名用户"
```

#### not 运算符
```python
# 基本用法
print(not True)   # False
print(not False)  # True

# 实际应用
is_logged_in = False
if not is_logged_in:
    print("请先登录")

# 双重否定
print(not not True)  # True

# 与其他运算符结合
x = 10
print(not x > 5)  # False
```

## 5. 位运算符

### 5.1 位运算符表

| 运算符 | 名称 | 示例 | 说明 |
|--------|------|------|------|
| & | 按位与 | x & y | 两位都为1时结果为1 |
| \| | 按位或 | x \| y | 至少一位为1时结果为1 |
| ^ | 按位异或 | x ^ y | 两位不同时结果为1 |
| ~ | 按位取反 | ~x | 0变1，1变0 |
| << | 左移 | x << 2 | 向左移动2位 |
| >> | 右移 | x >> 2 | 向右移动2位 |

### 5.2 详细说明

#### 按位与 (&)
```python
# 二进制操作
a = 12  # 1100
b = 10  # 1010
print(a & b)  # 8 (1000)

# 实际应用：判断奇偶
num = 7
if num & 1:  # 如果最后一位是1，则为奇数
    print("奇数")
else:
    print("偶数")
```

#### 按位或 (|)
```python
a = 12  # 1100
b = 10  # 1010
print(a | b)  # 14 (1110)

# 实际应用：权限管理
READ = 1    # 0001
WRITE = 2   # 0010
EXECUTE = 4 # 0100

permission = READ | WRITE  # 0011 = 3 (读+写权限)
print(permission)  # 3
```

#### 按位异或 (^)
```python
a = 12  # 1100
b = 10  # 1010
print(a ^ b)  # 6 (0110)

# 实际应用：交换变量（不用临时变量）
a = 5
b = 7
a = a ^ b
b = a ^ b
a = a ^ b
print(a, b)  # 7 5

# 实际应用：找出数组中唯一不重复的数
nums = [2, 3, 2, 4, 3]
result = 0
for num in nums:
    result ^= num
print(result)  # 4 (因为相同的数异或为0)
```

#### 按位取反 (~)
```python
a = 12
print(~a)  # -13 (补码运算：-(a+1))

# 原理：~x = -(x+1)
print(~0)   # -1
print(~-1)  # 0
print(~5)   # -6
```

#### 左移 (<<)
```python
a = 5  # 101
print(a << 1)  # 10 (1010) = 10
print(a << 2)  # 20 (10100) = 20

# 左移n位相当于乘以2^n
print(5 << 1)  # 10  (5 * 2^1)
print(5 << 2)  # 20  (5 * 2^2)
print(5 << 3)  # 40  (5 * 2^3)
```

#### 右移 (>>)
```python
a = 20  # 10100
print(a >> 1)  # 10 (1010) = 10
print(a >> 2)  # 5  (101) = 5

# 右移n位相当于除以2^n（向下取整）
print(20 >> 1)  # 10  (20 // 2^1)
print(20 >> 2)  # 5   (20 // 2^2)
print(20 >> 3)  # 2   (20 // 2^3)
```

## 6. 成员运算符

### 6.1 成员运算符表

| 运算符 | 说明 | 示例 |
|--------|------|------|
| in | 在序列中 | x in list |
| not in | 不在序列中 | x not in list |

### 6.2 详细说明

```python
# 列表成员检测
fruits = ["apple", "banana", "orange"]
print("apple" in fruits)      # True
print("grape" in fruits)      # False
print("grape" not in fruits)  # True

# 字符串成员检测
text = "Hello World"
print("Hello" in text)    # True
print("hello" in text)    # False (区分大小写)
print("xyz" not in text)  # True

# 字典成员检测（检测键）
person = {"name": "Alice", "age": 25}
print("name" in person)   # True
print("email" in person)  # False
print(25 in person)       # False (值不能用in检测)
print(25 in person.values())  # True (检测值需要用values())

# 元组成员检测
numbers = (1, 2, 3, 4, 5)
print(3 in numbers)       # True
print(10 not in numbers)  # True

# 集合成员检测（最快）
large_set = set(range(1000000))
print(999999 in large_set)  # True (O(1)时间复杂度)
```

## 7. 身份运算符

### 7.1 身份运算符表

| 运算符 | 说明 | 示例 |
|--------|------|------|
| is | 是同一个对象 | x is y |
| is not | 不是同一个对象 | x is not y |

### 7.2 详细说明

```python
# is vs ==
a = [1, 2, 3]
b = [1, 2, 3]
c = a

print(a == b)   # True (值相同)
print(a is b)   # False (不是同一个对象)
print(a is c)   # True (是同一个对象)

# None的检测（推荐用is）
x = None
print(x is None)  # True (推荐)
print(x == None)  # True (不推荐)

# 小整数和字符串的缓存
a = 256
b = 256
print(a is b)  # True (Python缓存了小整数-5到256)

a = 257
b = 257
print(a is b)  # False (超出缓存范围)

# 字符串驻留
s1 = "hello"
s2 = "hello"
print(s1 is s2)  # True (相同的字符串字面量会被复用)

s1 = "hello world"
s2 = "hello world"
print(s1 is s2)  # 可能True或False (取决于实现)
```

## 8. 运算符优先级

### 8.1 优先级表（从高到低）

| 优先级 | 运算符 | 说明 |
|--------|--------|------|
| 1 | ** | 幂运算 |
| 2 | ~, +, - | 按位取反、正号、负号 |
| 3 | *, /, //, % | 乘、除、整除、取模 |
| 4 | +, - | 加、减 |
| 5 | <<, >> | 左移、右移 |
| 6 | & | 按位与 |
| 7 | ^ | 按位异或 |
| 8 | \| | 按位或 |
| 9 | ==, !=, >, <, >=, <=, is, is not, in, not in | 比较运算符 |
| 10 | not | 逻辑非 |
| 11 | and | 逻辑与 |
| 12 | or | 逻辑或 |

### 8.2 优先级示例

```python
# 示例1：乘法优先于加法
result = 2 + 3 * 4
print(result)  # 14 (不是20)

# 示例2：幂运算优先于乘法
result = 2 * 3 ** 2
print(result)  # 18 (不是36)

# 示例3：使用括号改变优先级
result = (2 + 3) * 4
print(result)  # 20

# 示例4：复杂表达式
result = 10 + 5 * 2 ** 2 - 3
print(result)  # 10 + 5 * 4 - 3 = 10 + 20 - 3 = 27

# 示例5：逻辑运算符优先级
result = True or False and False
print(result)  # True (and优先于or)

result = (True or False) and False
print(result)  # False

# 示例6：比较运算符可以连续使用
x = 5
print(1 < x < 10)  # True (等价于 1 < x and x < 10)
print(1 < x < 3)   # False
```

## 9. 特殊用法和技巧

### 9.1 链式比较
```python
# 多重比较
age = 25
print(18 <= age <= 65)  # True

# 等价于
print(18 <= age and age <= 65)  # True

# 更多例子
x = 5
print(1 < x < 10 < 20)  # True
print(x == 5 == 5)      # True
```

### 9.2 三元运算符
```python
# 语法：value_if_true if condition else value_if_false
age = 20
status = "成年人" if age >= 18 else "未成年人"
print(status)  # "成年人"

# 嵌套三元运算符
score = 85
grade = "优秀" if score >= 90 else "良好" if score >= 80 else "及格" if score >= 60 else "不及格"
print(grade)  # "良好"
```

### 9.3 海象运算符 (:=) - Python 3.8+
```python
# 在表达式中赋值
numbers = [1, 2, 3, 4, 5]
if (n := len(numbers)) > 3:
    print(f"列表长度为{n}，超过3个元素")

# 在while循环中使用
while (line := input("输入（q退出）：")) != "q":
    print(f"你输入了：{line}")

# 列表推导式中使用
data = [1, 2, 3, 4, 5]
result = [y for x in data if (y := x * 2) > 4]
print(result)  # [6, 8, 10]
```

### 9.4 运算符重载
```python
# 自定义类中实现运算符
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __add__(self, other):
        """重载 + 运算符"""
        return Point(self.x + other.x, self.y + other.y)
    
    def __str__(self):
        return f"Point({self.x}, {self.y})"

p1 = Point(1, 2)
p2 = Point(3, 4)
p3 = p1 + p2
print(p3)  # Point(4, 6)
```
