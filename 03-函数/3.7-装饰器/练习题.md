## 装饰器原理

### 练习题

**题目 1：拆解 @**
```python
@deco
def f(): pass
```
等价于什么？

**题目 2：手动装饰**
不用 @，手动用装饰器 `log` 装饰函数 `add`，给出写法。

**题目 3：执行顺序**
```python
def deco(func):
    print(f"装饰 {func.__name__}")
    def wrapper(*a, **k):
        print(f"调用 {func.__name__}")
        return func(*a, **k)
    return wrapper

@deco
def add(a, b): return a + b

print("----")
print(add(1, 2))
```
打印顺序？

**题目 4：什么打印在定义时**
定义 `@deco def f(): pass` 时，会立刻调用 `print` 吗？

### 参考答案

<details>
<summary>点击查看答案</summary>

```python
# 题目 1
# 等价于：
# def f(): pass
# f = deco(f)

# 题目 2
def log(func):
    def wrapper(*a, **k):
        print("call", func.__name__)
        return func(*a, **k)
    return wrapper

def add(a, b): return a + b
add = log(add)
print(add(1, 2))   # call add \n 3

# 题目 3
def deco(func):
    print(f"装饰 {func.__name__}")
    def wrapper(*a, **k):
        print(f"调用 {func.__name__}")
        return func(*a, **k)
    return wrapper

@deco
def add(a, b): return a + b

print("----")
print(add(1, 2))
# 输出顺序：
# 装 add      <- 装饰器在 def 时立即应用
# ----
# 调用 add
# 3

# 题目 4
# 是的。装饰器在函数定义时（即读到 @deco + def 时）就立即执行，
# 包括装饰器体里的代码（除 wrapper 外）。
# wrapper 是 add 被调用时才执行。
```

### 练习题

**题目 1：日志装饰器**

**题目 2：trace 装饰器**

**题目 3：计时装饰器**

**题目 4：可选保留参数透传**
装饰任意参数的函数 `f(a, b=2, *c, d)`，调用 `f(1, 2, 3, 4, d=5)`，确认透传无误。

```python
import time
from functools import wraps

# 题目 1
def log(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"调用 {func.__name__}")
        result = func(*args, **kwargs)
        print(f"{func.__name__} 返回 {result}")
        return result
    return wrapper

@log
def add(a, b): return a + b

print(add(2, 3))
# 调用 add
# add 返回 5
# 5

# 题目 2
def trace(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"trace: {func.__name__}({args}, {kwargs})")
        result = func(*args, **kwargs)
        print(f"  -> {result}")
        return result
    return wrapper

@trace
def greet(name, greeting="Hi"): return f"{greeting}, {name}!"

print(greet("Alice", greeting="Yo"))
# trace: greet(('Alice',), {'greeting': 'Yo'})
#   -> Yo, Alice!
# Yo, Alice!

# 题目 3
def timer(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        t0 = time.perf_counter()
        result = func(*args, **kwargs)
        print(f"{func.__name__} 耗时 {time.perf_counter() - t0:.6f}s")
        return result
    return wrapper

@timer
def slow():
    time.sleep(0.1)
    return "done"

print(slow())
# slow 耗时 ~0.100003s
# done

# 题目 4
def deco(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

@deco
def f(a, b=2, *c, d):
    return (a, b, c, d)

print(f(1, 2, 3, 4, d=5))   # (1, 2, (3, 4), 5)
```

**题目 2：加 @wraps**

**题目 3：__wrapped__**
带 wraps 的 wrapper 有 `__wrapped__` 属性，验证可获取原 hello。

**题目 4：inspect.signature**
说明 `inspect.signature` 对未加 wraps 和加了 wraps 的 wrapper 的输出差异。

### 参考答案

<details>
<summary>点击查看答案</summary>

```python
import inspect
from functools import wraps

# 题目 1
def deco(func):
    def wrapper(*a, **k): return func(*a, **k)
    return wrapper

@deco
def hello(): """hi"""

print(hello.__name__)   # wrapper
print(hello.__doc__)    # None
# 问题：装饰后丢失原函数元信息，调试、文档生成会出错。

# 题目 2
def deco_good(func):
    @wraps(func)
    def wrapper(*a, **k):
        return func(*a, **k)
    return wrapper

@deco_good
def hello(): """hi"""

print(hello.__name__)   # hello
print(hello.__doc__)    # hi

# 题目 3
print(hello.__wrapped__)      # <function hello at ...>
print(hello.__wrapped__.__name__)  # hello

# 题目 4
# 未加 wraps：
# inspect.signature(wrapper) 显示 (*args, **kwargs) - wrapper 的签名
#
# 加了 wraps 后：
# inspect.signature(wrapper) 显示原函数的签名 (functools.wraps 让 inspect 通过 __wrapped__ 反推出)
def deco_no_wraps(func):
    def w(*a, **k): return func(*a, **k)
    return w

@deco_no_wraps
def fn(a, b, c): return a + b + c

print(inspect.signature(fn))   # (*args, **kwargs)

@deco_good
def fn2(a, b, c): return a + b + c
print(inspect.signature(fn2))  # (a, b, c) - 通过 __wrapped__
```

## 带参数装饰器

### 练习题

**题目 1：repeat**
实现 `@repeat(n)`，让函数调用 n 次，返回最后结果。验证 `@repeat(3)` 让 hello 打印 3 次。

**题目 2：level 日志**
实现 `@log(level="INFO")`，调用时打印 `[INFO] funcname`。

**题目 3：retry(times)**
实现 `@retry(times=3)`：捕获任何异常，最多重试 times 次，全失败抛出最后异常。

**题目 4：双层等价**
将 `@repeat(3) def f() = ...` 拆解为不带 @ 的等价写法。

### 参考答案

<details>
<summary>点击查看答案</summary>

```python
from functools import wraps

# 题目 1
def repeat(n):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            result = None
            for _ in range(n):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(3)
def hello():
    print("hi")

hello()
# hi
# hi
# hi

# 题目 2
def log(level="INFO"):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            print(f"[{level}] {func.__name__}")
            return func(*args, **kwargs)
        return wrapper
    return decorator

@log(level="DEBUG")
def go(): print("go")
go()   # [DEBUG] go

# 题目 3
def retry(times=3):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exc = None
            for i in range(times):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last_exc = e
                    print(f"第 {i+1} 失败: {e}")
            raise last_exc
        return wrapper
    return decorator

import random

@retry(times=5)
def flaky():
    if random.random() < 0.5:
        raise RuntimeError("炸")
    return "ok"

try:
    print(flaky())
except Exception as e:
    print("最终失败:", e)

# 题目 4
# @repeat(3) def f(): ...
# 等价：
# def f(): ...
# f = repeat(3)(f)
# 即先 repeat(3) 得到 decorator，再 decorator(f) 得到 wrapper
def repeat3_def(): pass
repeat3_def = repeat(3)(repeat3_def)
print(repeat3_def.__name__)  # wrapper (建议加 wraps 取得 repeat3_def)
```

</details>

## 类装饰器

### 练习题

**题目 1：CountCalls 类**
用类实现装饰器 `CountCalls`，统计函数被调用次数，可通过 `func.count` 查询。

**题目 2：用类装饰器装饰类**
实现 `add_repr(cls)` 给类加上 `__repr__` 输出属性字典。

**题目 3：singleton 类装饰器**
用类装饰器实现单例 `singleton(cls)`。

**题目 4：__call__ 检查**
验证类装饰器装饰后，原函数名是 `CountCalls` 的实例，但 `callable()` 仍 True。

### 参考答案

<details>
<summary>点击查看答案</summary>

```python
from functools import wraps

# 题目 1
class CountCalls:
    def __init__(self, func):
        self.func = func
        self.count = 0
        wraps(func)(self)

    def __call__(self, *args, **kwargs):
        self.count += 1
        return self.func(*args, **kwargs)

@CountCalls
def hello(name): return f"hi {name}"

print(hello("A"))
print(hello("B"))
print(hello.count)   # 2

# 题目 2
def add_repr(cls):
    def __repr__(self):
        return f"{cls.__name__}({self.__dict__})"
    cls.__repr__ = __repr__
    return cls

@add_repr
class P:
    def __init__(self, x, y):
        self.x = x; self.y = y

print(P(1, 2))   # P({'x': 1, 'y': 2})

# 题目 3
def singleton(cls):
    instances = {}
    @wraps(cls)
    def get(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    return get

@singleton
class Config:
    def __init__(self):
        self.x = 0

a = Config()
b = Config()
print(a is b)   # True

# 题目 4
print(type(hello))                # <class '__main__.CountCalls'>
print(isinstance(hello, CountCalls))  # True
print(callable(hello))            # True - 因为有 __call__
```

</details>

## 多装饰器叠加

### 练习题

**题目 1：洋葱顺序**
```python
def a(f):
    @wraps(f)
    def w(*args, **k):
        print("A前"); r = f(*args, **k); print("A后"); return r
    return w
def b(f):
    @wraps(f)
    def w(*args, **k):
        print("B前"); r = f(*args, **k); print("B后"); return r
    return w

@a
@b
def f(): print("f")
```
调用 f() 打印什么？

**题目 2：等价手动**
把上面 `@a @b def f` 拆解成不带 @ 的形式。

**题目 3：装饰器顺序影响**
说明 `@cached @logged` 与 `@logged @cached` 有何不同。

### 参考答案

<details>
<summary>点击查看答案</summary>

```python
from functools import wraps

# 题目 1
def a(f):
    @wraps(f)
    def w(*args, **k):
        print("A前"); r = f(*args, **k); print("A后"); return r
    return w
def b(f):
    @wraps(f)
    def w(*args, **k):
        print("B前"); r = f(*args, **k); print("B后"); return r
    return w

@a
@b
def f(): print("f")

f()
# 输出：
# A前
# B前
# f
# B后
# A后
# 洋葱模型：a 在最外层先入最后出。

# 题目 2
# @a @b def f(): ...
# 等价：
# def f(): ...
# f = b(f)
# f = a(f)
# 注意：先 b 再 a，因为 b 是离 f 最近的，先装饰底层
def f_raw(): print("f")
f_temp = b(f_raw)
f_final = a(f_temp)
f_final()

# 题目 3
# @cached @logged：cached 在外层，会缓存"已记录第几次"的结果
#   再调用时直接取缓存，不会再次走 logged 装饰，所以不会再次打印 "调用"
# @logged @cached：logged 在外层，每次调用都打印"调用"；
#   cached 在内层，命中后不重新计算
# 总体上：外层装饰器先执行、每次都执行；内层命中后可能直接跳过。
```

## 实用装饰器

### 练习题

**题目 1：deprecated**
实现 `@deprecated`，调用时用 `warnings.warn` 提示 DeprecationWarning。

**题目 2：require_admin**
实现权限装饰器 `@require_admin`，方法第一个参数是 user，user.is_admin 为假抛 PermissionError。

**题目 3：rate_limit**
实现 `@rate_limit(calls=3, period=1)`，限制每秒最多调用 calls 次，超限抛错。

**题目 4：memoize**
手写一个 memoize 装饰器（基于 dict），并用它实现 fib。

### 参考答案

<details>
<summary>点击查看答案</summary>

```python
import warnings
import time
from functools import wraps

# 题目 1
def deprecated(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        warnings.warn(f"{func.__name__} 已废弃", DeprecationWarning, stacklevel=2)
        return func(*args, **kwargs)
    return wrapper

@deprecated
def old_func():
    return "old"

with warnings.catch_warnings(record=True) as w:
    warnings.simplefilter("always")
    print(old_func())
    print("警告数:", len(w), w[0].category.__name__ if w else "无")

# 题目 2
def require_admin(func):
    @wraps(func)
    def wrapper(user, *args, **kwargs):
        if not getattr(user, "is_admin", False):
            raise PermissionError("需要管理员权限")
        return func(user, *args, **kwargs)
    return wrapper

class User:
    def __init__(self, name, is_admin=False):
        self.name, self.is_admin = name, is_admin

@require_admin
def delete(user, uid):
    return f"{user.name} 删除了 {uid}"

print(delete(User("admin", True), 1))
# delete(User("guest", False), 1)  # PermissionError

# 题目 3
def rate_limit(calls, period):
    history = []
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            now = time.time()
            history[:] = [t for t in history if now - t < period]
            if len(history) >= calls:
                raise RuntimeError("rate limit 超限")
            history.append(now)
            return func(*args, **kwargs)
        return wrapper
    return decorator

@rate_limit(calls=3, period=1)
def ping():
    return "pong"

print(ping(), ping(), ping())
try:
    ping()
except RuntimeError as e:
    print("限制", e)

# 题目 4
def memoize(func):
    cache = {}
    @wraps(func)
    def wrapper(*args):
        if args not in cache:
            cache[args] = func(*args)
        return cache[args]
    return wrapper

@memoize
def fib(n):
    return n if n < 2 else fib(n-1) + fib(n-2)

print(fib(40))   # 102334155, 瞬间
```

</details>

```python
import time
import warnings

from dataclasses import dataclass

# 综合题 1
def log_call(func):
    @wraps(func)
    def wrapper(self, *args, **kwargs):
        print(f"{self.__class__.__name__}.{func.__name__} 调用")
        return func(self, *args, **kwargs)
    return wrapper

class MyService:
    @log_call
    def serve(self):

# 综合题 2
def validate(**expected):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            for name, val in bound.arguments.items():
                if name in expected and not isinstance(val, expected[name]):
                    raise TypeError(f"{name} 应为 {expected[name]}，实际 {type(val).__name__}")
            return func(*args, **kwargs)
        return wrapper
    return decorator

@validate(x=int, y=int)
def add(x, y):
    return x + y

# add("1", 2)      # TypeError

def retry(times):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last = None
            for _ in range(times):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last = e
            raise last
        return wrapper
    return decorator

@retry(3)
@lru_cache()
def slow(x):
    print("computing", x)
    if x < 0:
        raise ValueError("负")
    return x * 2

print(slow(5))   # computing 5 \n 10
print(slow(5))   # 10  - 缓存命中，不重试也不再 computing

# 综合题 5
ROUTES = {}
def route(path):
    def decorator(func):
        ROUTES[path] = func
        return func
    return decorator

@route("/")
def index(): return "home"

@route("/about")
def about(): return "about"

print(ROUTES)
# {'/': <function index ...>, '/about': <function about ...>}
print(ROUTES["/"]())    # home
```

## 小结

- 装饰器：接收函数返回函数；`@deco` 等价 `f = deco(f)`
- 包装器用 `*args/**kwargs` 透传，且必须 `return func(...)`
- 必加 `@wraps(func)` 保留 `__name__/__doc__/签名`
- 带参数装饰器：三层嵌套（参数 → 装饰器 → wrapper）
- 类装饰器用 `__call__`，适合带状态
- 内置好用：`lru_cache / singledispatch / total_ordering / dataclass / property`
- 多装饰器从下往上应用，洋葱执行
- 多个装饰器顺序会影响结果，特别是缓存相关的位置
